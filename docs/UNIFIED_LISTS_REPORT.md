## 🔧 Техническая архитектура

### Структура сортировки
```typescript
// В каждом списке
const sortBy = ref('')
const sortOrder = ref<'asc' | 'desc'>('asc')

function handleSort(key: string) {
  if (sortBy.value === key) {
    sortOrder.value = sortOrder.value === 'asc' ? 'desc' : 'asc'
  } else {
    sortBy.value = key
    sortOrder.value = 'asc'
  }
  
  const ordering = sortOrder.value === 'desc' ? `-${key}` : key
  store.setFilters({ ordering })
}
```

### HTML структура таблицы
```html
<table class="modern-table">
  <thead>
    <tr>
      <th @click="handleSort('id')" class="cursor-pointer hover:bg-gray-50">
        ID
        <span v-if="sortBy === 'id'" class="ml-1">
          {{ sortOrder === 'asc' ? '↑' : '↓' }}
        </span>
      </th>
      <!-- Другие колонки -->
    </tr>
  </thead>
  <tbody>
    <!-- Данные -->
  </tbody>
</table>
```
