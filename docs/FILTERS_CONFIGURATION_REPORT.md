## 🔧 Технические детали

### Автоматические фильтры
```typescript
// В каждом списке
const debouncedSearch = debounce(() => {
  store.fetchList() // или reload(1)
}, 500)

watch(
  () => store.filters, // или filters.value
  () => {
    store.pagination.page = 1 // или page.value = 1
    debouncedSearch()
  },
  { deep: true }
)
```

### Сортировка заголовков
```typescript
function handleSort(key: string) {
  if (sortBy.value === key) {
    sortOrder.value = sortOrder.value === 'asc' ? 'desc' : 'asc'
  } else {
    sortBy.value = key
    sortOrder.value = 'asc'
  }
  
  const ordering = sortOrder.value === 'desc' ? `-${key}` : key
  store.setFilters({ ordering }) // или filters.value.ordering = ordering
}
```

### HTML структура заголовков
```html
<th @click="handleSort('id')" class="cursor-pointer hover:bg-gray-50">
  ID
  <span v-if="sortBy === 'id'" class="ml-1">
    {{ sortOrder === 'asc' ? '↑' : '↓' }}
  </span>
</th>
```
